"use client";

// Project Budget panel — the "where are we at" view for the Money tab.
// Pure render: takes a BudgetView (lib/budget-types.ts), no actions, no fetch.
// Built to sit inside the project page like MoneyPanel does: no page header,
// Card/Chip/Eyebrow primitives, Tailwind tokens from app/globals.css.
//
// Layout (top to bottom): budget bar + three KPIs, cost-by-trade chart with
// over/under lists, who-pays chart, change-order chart, then <details>
// drill-downs (lines table, change orders, funding, ledger, notes).

import { useState } from "react";
import { ChevronDown } from "lucide-react";
import { Card, Chip, Eyebrow } from "@/components/ui";
import type { ChipKind } from "@/components/ui/Chip";
import { fmtUsd } from "@/lib/cost-book-units";
import {
  computeTotals,
  coCredit,
  fmtK,
  fmtSigned,
  lineVariance,
  type BudgetView,
  type BudgetLine,
  type BudgetCoStatus,
} from "@/lib/budget-types";

const CO_CHIP: Record<BudgetCoStatus, ChipKind> = {
  draft: "ghost", sent: "accent", approved: "money", declined: "flag", billed: "info", paid: "money",
};
const CO_LABEL: Record<BudgetCoStatus, string> = {
  draft: "Draft", sent: "Awaiting signature", approved: "Approved", declined: "Declined", billed: "Billed", paid: "Paid",
};
const INV_CHIP: Record<"draft" | "sent" | "paid", ChipKind> = { draft: "ghost", sent: "accent", paid: "money" };
const SUB_CHIP: Record<"quote" | "submitted" | "approved" | "paid", ChipKind> = {
  quote: "ai", submitted: "accent", approved: "info", paid: "money",
};

const pct = (a: number, b: number) => (b > 0 ? Math.max(0, Math.min(100, (a / b) * 100)) : 0);

// Colors read from the app's CSS variables so the SVGs follow the theme.
const C = {
  spent: "var(--accent)", comm: "var(--ai)", est: "var(--paper-4)", over: "var(--flag)",
  under: "var(--money)", owner: "var(--info)", funder: "var(--money)", cost: "var(--accent)",
  co: "var(--info)", credit: "var(--paper-4)", ink: "var(--ink)", ink2: "var(--ink-2)", ink3: "var(--ink-3)",
  rule: "var(--rule-soft)", paper: "var(--paper)",
};

export function BudgetPanel({ budget }: { budget: BudgetView }) {
  const t = computeTotals(budget);
  const over = t.headroomCents < 0;
  const ownerPay = t.paidBy.find((p) => p.isOwner);
  const active = budget.lines.filter((l) => !l.credited);
  const tradeLines = active.filter((l) => l.kind === "trade");

  return (
    <div className="flex flex-col gap-4">
      {/* ---------- Hero: budget bar ---------- */}
      <Card className="p-4 sm:p-5">
        <div className="flex flex-wrap items-baseline justify-between gap-x-4 gap-y-1">
          <div className="font-serif text-[30px] leading-none text-ink sm:text-[34px]">
            {fmtUsd(t.budgetCents)}
            <span className="ml-2 font-sans text-[12px] text-ink-3">{budget.budgetLabel}</span>
          </div>
          <div className={`font-serif text-[20px] ${over ? "text-flag" : "text-money"}`}>
            {fmtK(Math.abs(t.headroomCents))} {over ? "over" : "headroom"}
          </div>
        </div>
        {budget.budgetCaption && <div className="mt-1 text-[12px] text-ink-3">{budget.budgetCaption}</div>}

        <div className="mt-3.5 flex h-10 overflow-hidden rounded-md bg-paper-3">
          <div className="h-full bg-accent" style={{ width: `${pct(t.spentCents, t.budgetCents)}%` }} title="Spent" />
          <div className="h-full bg-ai" style={{ width: `${pct(t.committedCents, t.budgetCents)}%` }} title="Committed" />
          <div
            className="h-full"
            style={{
              width: `${pct(t.estToFinishCents, t.budgetCents)}%`,
              backgroundImage: "repeating-linear-gradient(135deg, var(--paper-4) 0 7px, var(--paper-3) 7px 14px)",
            }}
            title="Estimated to finish"
          />
          {over && <div className="h-full bg-flag" style={{ width: `${pct(-t.headroomCents, t.budgetCents)}%` }} />}
        </div>
        <div className="mt-2.5 flex flex-wrap gap-x-5 gap-y-1.5 text-[12.5px] text-ink-2">
          <Legend swatch="bg-accent" label="Spent" value={fmtK(t.spentCents)} />
          <Legend swatch="bg-ai" label="Committed" value={fmtK(t.committedCents)} />
          <Legend swatch="bg-paper-4 border border-rule" label="Est. to finish" value={fmtK(t.estToFinishCents)} />
          <Legend swatch={over ? "bg-flag" : "bg-paper-3 border border-rule"} label={over ? "Over" : "Headroom"} value={fmtK(Math.abs(t.headroomCents))} />
        </div>
      </Card>

      <div className="grid grid-cols-2 gap-2.5 sm:grid-cols-3">
        <Kpi label="Spent and collected" value={fmtK(t.spentCents)} note={`${pct(t.spentCents, t.budgetCents).toFixed(0)}% of budget`} />
        <Kpi label="Budget left to draw" value={fmtK(t.leftToDrawCents)} note={`Est. to finish ${fmtK(t.estToFinishCents)}`} />
        {ownerPay && (
          <Kpi
            label={`${ownerPay.label} pays, projected`}
            value={fmtK(ownerPay.amountCents)}
            note={[
              t.coOwnerCents ? `incl. ${fmtK(t.coOwnerCents)} change orders` : null,
              over ? `incl. ${fmtK(-t.headroomCents)} overrun` : null,
              budget.retainageCents ? `${fmtK(budget.retainageCents)} retainage held` : null,
            ].filter(Boolean).join(" · ") || "Base scope share"}
          />
        )}
      </div>

      {/* ---------- Cost by trade ---------- */}
      <Section title="Cost by trade" sub="Bars are spent, committed, and estimated to finish. The outline is the budget for that trade.">
        <Card className="overflow-x-auto p-4">
          <TradeChart lines={tradeLines} />
          <div className="mt-2 flex flex-wrap gap-x-4 gap-y-1 text-[12px] text-ink-2">
            <Legend swatch="bg-accent" label="Spent" />
            <Legend swatch="bg-ai" label="Committed" />
            <Legend swatch="bg-paper-4 border border-rule" label="Est. to finish" />
            <Legend swatch="border border-ink-3 bg-transparent" label="Budget" />
          </div>
        </Card>
        <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-2">
          <VarianceList title="Over budget" rows={t.overLines} tone="flag" footLabel="Total over" footValue={t.overLines.reduce((s, x) => s + x.varianceCents, 0)} />
          <VarianceList title="Under budget" rows={t.underLines} tone="money"
            footLabel={t.creditedCents ? "Budget moved to change orders" : "Total under"}
            footValue={t.creditedCents ? t.creditedCents : -t.underLines.reduce((s, x) => s + x.varianceCents, 0)} />
        </div>
      </Section>

      {/* ---------- Who pays ---------- */}
      <Section title="Who pays for what" sub="Total project cost against where the money comes from.">
        <Card className="overflow-x-auto p-4">
          <PayChart t={t} />
        </Card>
      </Section>

      {/* ---------- Change orders ---------- */}
      {budget.changeOrders.length > 0 && (
        <Section title="Change orders" sub="Work added beyond the base scope. Each one credits back any base budget it replaces. Pending orders are outlined.">
          <Card className="overflow-x-auto p-4">
            <CoChart budget={budget} ownerLabel={ownerPay?.label ?? "Client"} />
          </Card>
        </Section>
      )}

      {/* ---------- Drill-downs ---------- */}
      <Section title="Details">
        <Fold title="Base scope, line by line" right={`${active.length} lines · ${fmtK(t.projectedCents)} projected`}>
          <LinesTable budget={budget} t={t} />
        </Fold>
        {budget.changeOrders.length > 0 && (
          <Fold title="Change order detail" right={`${budget.changeOrders.length} orders · ${fmtK(t.coOwnerCents)} ${ownerPay?.label.toLowerCase() ?? "client"}`}>
            <div className="grid grid-cols-1 gap-2.5 sm:grid-cols-2 lg:grid-cols-3">
              {budget.changeOrders.map((c) => {
                const credit = coCredit(c);
                return (
                  <Card key={c.id} kind="soft" className="flex flex-col gap-2 p-3.5">
                    <div>
                      <Eyebrow className="text-info">{c.id}</Eyebrow>
                      <div className="font-serif text-[17px] leading-tight text-ink">{c.title}</div>
                    </div>
                    {c.description && <p className="text-[12.5px] text-ink-2">{c.description}</p>}
                    <div className="grid grid-cols-[1fr_auto] gap-x-3 gap-y-0.5 border-t border-rule-soft pt-2 text-[12.5px]">
                      <span>Total</span><span className="font-mono">{fmtUsd(c.totalCents)}</span>
                      <span>Base budget credit</span><span className="font-mono">{credit ? `-${fmtUsd(credit)}` : "none"}</span>
                      <span className="font-semibold">{c.paidBy === "funder" ? "Funder pays" : `${ownerPay?.label ?? "Client"} pays`}</span>
                      <span className="font-mono font-semibold">{fmtUsd(c.totalCents - credit)}</span>
                      <span>Billed</span><span className="font-mono">{fmtUsd(c.billedCents)}</span>
                    </div>
                    <div className="flex flex-wrap items-center gap-2">
                      <Chip kind={CO_CHIP[c.status]} dot>{CO_LABEL[c.status]}</Chip>
                      {c.vendor && <span className="text-[11.5px] text-ink-3">{c.vendor}</span>}
                    </div>
                  </Card>
                );
              })}
            </div>
          </Fold>
        )}
        {budget.fundingEvents.length > 0 && (
          <Fold title="Funding" right={`${budget.fundingEvents.length} events`}>
            <table className="w-full text-[12.5px]">
              <thead><tr className="text-left"><Th>Source</Th><Th right>Amount</Th><Th>Trigger</Th><Th>Status</Th></tr></thead>
              <tbody>
                {budget.fundingEvents.map((f, i) => (
                  <tr key={i} className="border-t border-rule-soft">
                    <Td>{f.source}</Td>
                    <Td right mono>{fmtUsd(f.amountCents)}</Td>
                    <Td muted>{f.trigger}</Td>
                    <Td><Chip kind={f.status === "received" ? "money" : f.status === "requested" ? "accent" : "ghost"} dot>{f.statusLabel ?? f.status}</Chip></Td>
                  </tr>
                ))}
              </tbody>
            </table>
          </Fold>
        )}
        <Fold title="Ledger" right={`${budget.clientInvoices.length} client · ${budget.subInvoices.length} sub`}>
          <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
            <table className="w-full text-[12.5px]">
              <thead><tr className="text-left"><Th>Inv</Th><Th>Date</Th><Th>Items</Th><Th right>Amount</Th><Th>Status</Th></tr></thead>
              <tbody>
                {budget.clientInvoices.map((i) => (
                  <tr key={i.number} className="border-t border-rule-soft">
                    <Td mono>{i.number}</Td><Td muted nowrap>{i.dateLabel}</Td><Td>{i.label}</Td>
                    <Td right mono>{fmtUsd(i.amountCents)}</Td>
                    <Td><Chip kind={INV_CHIP[i.status]} dot>{i.statusLabel}</Chip></Td>
                  </tr>
                ))}
                {budget.clientInvoices.length === 0 && <tr><Td muted>No client invoices yet.</Td></tr>}
              </tbody>
            </table>
            <table className="w-full text-[12.5px]">
              <thead><tr className="text-left"><Th>Sub / vendor</Th><Th>Date</Th><Th right>Amount</Th><Th>Note</Th></tr></thead>
              <tbody>
                {budget.subInvoices.map((s, i) => (
                  <tr key={i} className="border-t border-rule-soft">
                    <Td>{s.vendor}</Td><Td muted nowrap>{s.dateLabel}</Td>
                    <Td right mono>{fmtUsd(s.amountCents)}</Td>
                    <Td><Chip kind={SUB_CHIP[s.status]}>{s.note ?? s.status}</Chip></Td>
                  </tr>
                ))}
                {budget.subInvoices.length === 0 && <tr><Td muted>No sub invoices yet.</Td></tr>}
              </tbody>
            </table>
          </div>
        </Fold>
        {budget.notes && budget.notes.length > 0 && (
          <Fold title="Notes and open questions" right={`${budget.notes.length}`}>
            <ul className="list-disc space-y-1 pl-5 text-[12.5px] text-ink-2">
              {budget.notes.map((n, i) => <li key={i}>{n}</li>)}
            </ul>
          </Fold>
        )}
      </Section>
      <div className="text-right font-mono text-[10px] text-ink-4">as of {budget.asOfLabel}</div>
    </div>
  );
}

function TradeChart({ lines }: { lines: BudgetLine[] }) {
  const rows = [...lines].sort(
    (a, b) => Math.max(b.budgetCents, b.spentCents + b.committedCents + b.estToFinishCents)
            - Math.max(a.budgetCents, a.spentCents + a.committedCents + a.estToFinishCents),
  );
  if (!rows.length) return <div className="text-[13px] text-ink-3">No trade lines yet.</div>;
  const W = 860, labW = 180, valW = 96, rowH = 32, top = 6;
  const maxV = Math.max(1, ...rows.map((r) => Math.max(r.budgetCents, r.spentCents + r.committedCents + r.estToFinishCents)));
  const plotW = W - labW - valW - 8;
  const sx = (v: number) => (v / maxV) * plotW;
  return (
    <svg viewBox={`0 0 ${W} ${top + rows.length * rowH}`} className="block min-w-[640px] w-full" style={{ fontFamily: "var(--font-sans)" }}>
      {rows.map((r, i) => {
        const y = top + i * rowH, h = 17, x0 = labW;
        const v = lineVariance(r);
        let x = x0;
        const segs: [string, number][] = [[C.spent, r.spentCents], [C.comm, r.committedCents], [C.est, r.estToFinishCents]];
        const vx = Math.max(x0 + sx(r.spentCents + r.committedCents + r.estToFinishCents), x0 + sx(r.budgetCents)) + 8;
        return (
          <g key={r.id}>
            <text x={labW - 10} y={y + h / 2 + 4} textAnchor="end" fontSize="12" fontWeight={600} fill={C.ink}>{r.trade}</text>
            {segs.map(([fill, val], k) => {
              if (val <= 0) return null;
              const el = <rect key={k} x={x} y={y} width={sx(val)} height={h} fill={fill} />;
              x += sx(val);
              return el;
            })}
            {r.budgetCents > 0 && <rect x={x0} y={y - 2} width={sx(r.budgetCents)} height={h + 4} rx={2} fill="none" stroke={C.ink3} strokeWidth={1.5} />}
            <text x={vx} y={y + h / 2 + 4} fontSize="11" fontFamily="var(--font-mono)" fill={Math.abs(v) < 100 ? C.ink3 : v > 0 ? C.over : C.under}>
              {Math.abs(v) < 100 ? "on budget" : fmtSigned(v)}
            </text>
          </g>
        );
      })}
    </svg>
  );
}

function PayChart({ t }: { t: ReturnType<typeof computeTotals> }) {
  const W = 860, lab = 130, plot = W - lab - 16, bh = 32;
  const mx = Math.max(1, t.projectTotalCents);
  const sx = (v: number) => (v / mx) * plot;
  const Stack = ({ y, label, segs }: { y: number; label: string; segs: [string, number, string][] }) => {
    let x = lab;
    return (
      <g>
        <text x={lab - 10} y={y + bh / 2 + 4} textAnchor="end" fontSize="12" fontWeight={600} fill={C.ink}>{label}</text>
        {segs.map(([fill, v, name], i) => {
          if (v <= 0) return null;
          const w = sx(v), x1 = x; x += w;
          return (
            <g key={i}>
              <rect x={x1} y={y} width={w} height={bh} fill={fill} />
              {w > 90
                ? <text x={x1 + 8} y={y + bh / 2 + 4} fontSize="12" fontWeight={600} fill={C.paper}>{name} {fmtK(v)}</text>
                : <text x={x1 + w / 2} y={y + bh + 13} textAnchor="middle" fontSize="11" fontFamily="var(--font-mono)" fill={C.ink3}>{name} {fmtK(v)}</text>}
            </g>
          );
        })}
      </g>
    );
  };
  const payers: [string, number, string][] = t.paidBy.map((p) => [p.isOwner ? C.owner : C.funder, p.amountCents, p.label]);
  return (
    <svg viewBox={`0 0 ${W} 132`} className="block min-w-[640px] w-full">
      <Stack y={10} label="Project cost" segs={[[C.cost, t.projectedCents, "Base scope"], [C.co, t.coTotalCents, "Change orders"]]} />
      <Stack y={72} label="Paid by" segs={payers} />
      <text x={lab} y={126} fontSize="11" fontFamily="var(--font-mono)" fill={C.ink3}>
        Total {fmtUsd(t.projectTotalCents)}{t.coPendingCents ? ` · ${fmtK(t.coPendingCents)} in pending change orders not included` : ""}{t.unfundedCents ? ` · ${fmtK(t.unfundedCents)} unfunded` : ""}
      </text>
    </svg>
  );
}

function CoChart({ budget, ownerLabel }: { budget: BudgetView; ownerLabel: string }) {
  const cos = budget.changeOrders.filter((c) => c.status !== "declined");
  const W = 860, lab = 210, plot = W - lab - 120, rh = 38;
  const mx = Math.max(1, ...cos.map((c) => Math.abs(c.totalCents)));
  const sx = (v: number) => (Math.abs(v) / mx) * plot;
  return (
    <svg viewBox={`0 0 ${W} ${cos.length * rh + 8}`} className="block min-w-[640px] w-full">
      {cos.map((c, i) => {
        const y = 8 + i * rh, credit = coCredit(c), pending = c.status === "draft" || c.status === "sent";
        const net = c.totalCents - credit;
        return (
          <g key={c.id}>
            <text x={lab - 10} y={y + 14} textAnchor="end" fontSize="12" fontWeight={600} fill={C.ink}>{c.id} {c.title}</text>
            <rect x={lab} y={y} width={sx(net)} height={20} fill={pending ? "none" : c.totalCents < 0 ? C.under : C.co} stroke={pending ? C.co : "none"} strokeWidth={1.5} strokeDasharray={pending ? "4 3" : undefined} />
            {credit > 0 && <rect x={lab + sx(net)} y={y} width={sx(credit)} height={20} fill={C.credit} />}
            <text x={lab + sx(c.totalCents) + 8} y={y + 14} fontSize="11" fontFamily="var(--font-mono)" fill={C.ink3}>
              {fmtK(net)} {c.paidBy === "funder" ? "funder" : ownerLabel.toLowerCase()}{pending ? " · pending" : ""}
            </text>
          </g>
        );
      })}
    </svg>
  );
}

function LinesTable({ budget, t }: { budget: BudgetView; t: ReturnType<typeof computeTotals> }) {
  const dash = <span className="text-ink-4">–</span>;
  const other = budget.lines.filter((l) => !l.credited && l.kind !== "trade");
  return (
    <div className="overflow-x-auto">
      <table className="w-full min-w-[720px] text-[12.5px]">
        <thead><tr className="text-left"><Th>Line</Th><Th right>Budget</Th><Th right>Spent</Th><Th right>Committed</Th><Th right>Est. to finish</Th><Th right>Variance</Th><Th>Status</Th></tr></thead>
        <tbody>
          {budget.lines.map((l) => {
            const v = lineVariance(l);
            return (
              <tr key={l.id} className={`border-t border-rule-soft ${l.credited ? "text-ink-3" : ""}`}>
                <Td>
                  <div className="font-semibold" title={l.source}>{l.trade}{l.kind !== "trade" && <span className="ml-1.5 font-mono text-[9.5px] uppercase text-ink-4">{l.kind}</span>}</div>
                  {l.detail && <div className="text-[11.5px] text-ink-3">{l.detail}</div>}
                  {l.flags?.length ? <div className="mt-0.5 flex flex-wrap gap-1">{l.flags.map((f) => <Chip key={f} kind="flag">{f}</Chip>)}</div> : null}
                </Td>
                <Td right mono>{fmtUsd(l.budgetCents)}</Td>
                <Td right mono>{l.spentCents ? fmtUsd(l.spentCents) : dash}</Td>
                <Td right mono>{l.committedCents ? fmtUsd(l.committedCents) : dash}</Td>
                <Td right mono>{l.estToFinishCents ? fmtUsd(l.estToFinishCents) : dash}</Td>
                <Td right>
                  {l.credited ? <Chip kind="info">credited{l.creditedTo ? ` to ${l.creditedTo}` : ""}</Chip>
                    : Math.abs(v) < 100 ? <Chip kind="ghost">on budget</Chip>
                    : <Chip kind={v > 0 ? "flag" : "money"}>{fmtSigned(v)}</Chip>}
                </Td>
                <Td>{l.status && <Chip kind={l.statusKind ?? "ghost"} dot>{l.status}</Chip>}</Td>
              </tr>
            );
          })}
        </tbody>
        <tfoot className="bg-paper-2 font-semibold">
          <tr className="border-t border-rule">
            <Td>Budget total</Td>
            <Td right mono>{fmtUsd(t.budgetCents)}</Td>
            <Td right mono>{fmtUsd(t.spentCents)}</Td>
            <Td right mono>{fmtUsd(t.committedCents)}</Td>
            <Td right mono>{fmtUsd(t.estToFinishCents)}</Td>
            <Td right><Chip kind={t.headroomCents < 0 ? "flag" : "money"}>{fmtSigned(-t.headroomCents)}</Chip></Td>
            <Td />
          </tr>
        </tfoot>
      </table>
      {(t.creditedCents > 0 || other.length > 0) && (
        <p className="mt-2 text-[11.5px] text-ink-3">
          {t.creditedCents > 0 && <>Credited to change orders: {fmtUsd(t.creditedCents)}. </>}
          {other.length > 0 && <>Non-trade lines ({other.map((o) => o.trade).join(", ")}) total {fmtUsd(t.otherBudgetCents)} and are not drawn in the chart.</>}
        </p>
      )}
    </div>
  );
}

function VarianceList({ title, rows, tone, footLabel, footValue }: {
  title: string; rows: { line: BudgetLine; varianceCents: number }[]; tone: "flag" | "money"; footLabel: string; footValue: number;
}) {
  return (
    <Card className="p-3.5">
      <div className="mb-1 font-serif text-[15px] text-ink">{title}</div>
      {rows.length === 0 && <div className="py-1 text-[12.5px] text-ink-3">None</div>}
      {rows.map(({ line, varianceCents }) => (
        <div key={line.id} className="flex justify-between gap-3 border-t border-rule-soft py-1 text-[12.5px]">
          <span>{line.trade}</span>
          <span className={`font-mono text-${tone}`}>{fmtSigned(varianceCents)}</span>
        </div>
      ))}
      <div className="mt-0.5 flex justify-between gap-3 border-t border-rule pt-1 text-[12.5px] font-semibold">
        <span>{footLabel}</span><span className="font-mono">{fmtK(footValue)}</span>
      </div>
    </Card>
  );
}

function Fold({ title, right, children }: { title: string; right?: string; children: React.ReactNode }) {
  const [open, setOpen] = useState(false);
  return (
    <Card className="overflow-hidden p-0">
      <button type="button" onClick={() => setOpen((o) => !o)} className="flex w-full items-center justify-between gap-3 px-4 py-3 text-left hover:bg-paper-2">
        <span className="font-serif text-[16px] text-ink">{title}</span>
        <span className="flex items-center gap-2 font-mono text-[11px] text-ink-3">
          {right}
          <ChevronDown className={`size-3.5 transition-transform ${open ? "rotate-180" : ""}`} strokeWidth={1.75} />
        </span>
      </button>
      {open && <div className="border-t border-rule-soft px-4 py-3">{children}</div>}
    </Card>
  );
}

function Section({ title, sub, children }: { title: string; sub?: string; children: React.ReactNode }) {
  return (
    <div className="flex flex-col gap-2.5">
      <div>
        <h3 className="font-serif text-[18px] text-ink">{title}</h3>
        {sub && <div className="text-[12.5px] text-ink-3">{sub}</div>}
      </div>
      {children}
    </div>
  );
}

function Kpi({ label, value, note }: { label: string; value: string; note?: string }) {
  return (
    <Card className="px-3.5 py-3">
      <div className="text-[11.5px] text-ink-3">{label}</div>
      <div className="mt-0.5 font-serif text-[24px] leading-tight text-ink">{value}</div>
      {note && <div className="mt-0.5 text-[11.5px] text-ink-3">{note}</div>}
    </Card>
  );
}

function Legend({ swatch, label, value }: { swatch: string; label: string; value?: string }) {
  return (
    <span className="inline-flex items-center gap-1.5">
      <i className={`inline-block size-3 rounded-[3px] ${swatch}`} />
      {label}{value && <b className="font-mono font-medium">{value}</b>}
    </span>
  );
}

function Th({ children, right }: { children?: React.ReactNode; right?: boolean }) {
  return <th className={`px-2 pb-1.5 font-mono text-[9.5px] font-semibold uppercase tracking-[0.1em] text-ink-3 ${right ? "text-right" : ""}`}>{children}</th>;
}
function Td({ children, right, mono, muted, nowrap }: { children?: React.ReactNode; right?: boolean; mono?: boolean; muted?: boolean; nowrap?: boolean }) {
  return <td className={`px-2 py-2 align-top ${right ? "text-right" : ""} ${mono ? "font-mono text-[12px]" : ""} ${muted ? "text-ink-3" : ""} ${nowrap ? "whitespace-nowrap" : ""}`}>{children}</td>;
}
